if match:
    decoded = urllib.parse.unquote(match.group(1))
    parsed = json.loads(decoded)
    toc = parsed.get('book', {}).get('toc', [])
    with open('/tmp/toc.json', 'w') as f:
        json.dump(toc, f, ensure_ascii=False)
    print(f'TOC saved: {len(toc)} entries')
